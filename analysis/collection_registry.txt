# Which collections under output/ count as evaluation data — shared rules.
#
# This file holds only rules that are true for anyone running this pipeline,
# because the directories they match are ones the pipeline itself creates:
# sources/ writes ia_*/nypl_*/loc_* probes, the evidence scripts write harvest_*,
# and backup tooling writes _*. A fresh clone gets rules that actually fire.
#
# Judgements about *your* corpus — which volumes are manuscripts, which genres
# you were experimenting with — go in analysis/collection_registry.local.txt,
# which is gitignored. See collection_registry.local.txt.example. Both files are
# loaded, local second, and a local `include` overrides a shared `exclude`.
#
# Format:  <verb> <glob> <category> [# note]
#   exclude — keep out of cross-collection analysis
#   include — force back in, overriding any exclude
#
# Collections matching nothing are INCLUDED: production is the norm, and a new
# volume must not vanish from an evaluation because nobody registered it. The
# tools print what they excluded, so the opposite mistake stays visible — an
# unnoticed backup directory holding 6,862 aligned files and zero images is how
# a cross-collection table got polluted in the first place.

# --- infrastructure: not collections at all -------------------------------
exclude  _*          infrastructure   # _ocr_backups, _frontmatter

# --- source probes: a dozen pages pulled to check a IIIF endpoint works ----
exclude  ia_*        source-probe
exclude  nypl_*      source-probe
exclude  loc_*       source-probe

# --- evidence slices written by the harness under docs/evidence/ ----------
exclude  harvest_*   evidence-slice
